function requestLoan() { if(confirm('আপনি কি নিশ্চিতভাবে ১০০০ টাকা লোন নিতে চান? এটি আপনার My Cash-এ যুক্ত হবে।')) { // ব্যাকএন্ডে ডেটা পাঠানোর কোড (PHP বা Firebase-এর মাধ্যমে) fetch('process_loan.php', { method: 'POST', headers: { 'Content-Type': 'application/json', }, body: JSON.stringify({ action: 'accept_loan', amount: 1000 }) }) .then(response => response.json()) .then(data => { if(data.success) { alert('অভিনন্দন! লোন সফল হয়েছে এবং আপনার My Cash-এ টাকা যোগ করা হয়েছে।'); location.reload(); // পেজ রিফ্রেশ করে আপডেট দেখানোর জন্য } else { alert(data.message || 'লোন নিতে সমস্যা হয়েছে।'); } }) .catch(error => console.error('Error:', error)); } } import { db, auth } from './firebase-config.js'; // আপনার ফায়ারবেস কনফিগ ফাইল import { doc, getDoc, runTransaction, onSnapshot } from "https://www.gstatic.com/firebasejs/10.x.x/firebase-firestore.js"; // ইউজারের লোন স্ট্যাটাস চেক ও রিয়েলটাইম আপডেট function listenUserStatus(userId) { const userRef = doc(db, "users", userId); onSnapshot(userRef, (docSnap) => { if (docSnap.exists()) { const data = docSnap.data(); const loanStatus = data.loan_status || "none"; const currentCash = data.my_cash || 0; const titleEl = document.getElementById('statusTitle'); const descEl = document.getElementById('statusDesc'); const actionArea = document.getElementById('actionArea'); if (loanStatus === "none") { titleEl.innerText = "⚠️ কোনো লোন কার্যকর নেই"; descEl.innerText = "আপনার অ্যাকাউন্টে বর্তমানে কোনো লোন নেই। আপনি চাইলে ১,০০০ টাকা লোন নিতে পারেন।"; actionArea.style.display = "block"; } else if (loanStatus === "active") { titleEl.innerText = "✅ লোন কার্যকর রয়েছে"; descEl.innerText = `আপনার ${data.loan_amount || 1000} টাকার লোন চালু আছে। বর্তমান My Cash ব্যালেন্স: ৳${currentCash}`; actionArea.style.display = "none"; } } }); } // লোন একসেপ্ট করার ফাংশন window.takeLoan = async function() { const user = auth.currentUser; if (!user) { alert("দয়া করে প্রথমে লগইন করুন।"); return; } const loanAmount = 1000; // লোন এর পরিমাণ const userRef = doc(db, "users", user.uid); if (!confirm(`আপনি কি নিশ্চিত যে ${loanAmount} টাকা লোন নিতে চান?`)) return; try { await runTransaction(db, async (transaction) => { const userDoc = await transaction.get(userRef); if (!userDoc.exists()) { throw "ইউজার ডাটা পাওয়া যায়নি!"; } const userData = userDoc.data(); // লোন অলরেডি নেওয়া থাকলে আর নিতে দেওয়া হবে না if (userData.loan_status === "active") { throw "আপনার ইতিমধ্যে একটি লোন অ্যাক্টিভ আছে!"; } const currentCash = userData.my_cash || 0; const newCashBalance = currentCash + loanAmount; // ডাটাবেজ আপডেট transaction.update(userRef, { my_cash: newCashBalance, loan_status: "active", loan_amount: loanAmount, loan_date: new Date().toISOString() }); }); alert("অভিনন্দন! লোন সফলভাবে জমা হয়েছে এবং My Cash-এ যুক্ত করা হয়েছে।"); } catch (error) { console.error("Loan Transaction Failed: ", error); alert(error || "লোন প্রক্রিয়াটি সম্পন্ন করা সম্ভব হয়নি।"); } };
