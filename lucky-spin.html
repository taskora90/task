import { getFirestore, doc, getDoc, updateDoc, increment, collection, query, where, getDocs } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js"; // ইউজার যখন সফলভাবে প্যাকেজ কিনবে, তখন এই ফাংশনটি কল হবে async function onPackagePurchaseSuccess(currentUserUid) { const db = getFirestore(); try { // ১. বর্তমান ইউজারের ডাটা আনব, দেখতে সে কার রেফারে এসেছে (referredBy) const userRef = doc(db, "users", currentUserUid); const userSnap = await getDoc(userRef); if (userSnap.exists()) { const userData = userSnap.data(); const referrerCode = userData.referredBy; // যার কোড ব্যবহার করে অ্যাকাউন্ট খুলেছিল // ২. যদি কেউ রেফার করে থাকে, তবে রেফারারের খোঁজ করব if (referrerCode) { const usersRef = collection(db, "users"); const q = query(usersRef, where("referralCode", "==", referrerCode)); const querySnapshot = await getDocs(q); if (!querySnapshot.empty) { // রেফারার পাওয়া গেছে querySnapshot.forEach(async (referrerDoc) => { const referrerRef = doc(db, "users", referrerDoc.id); // ৩. রেফারারের লাকি স্পিন ১ বাড়িয়ে দেব await updateDoc(referrerRef, { luckySpinCount: increment(1) }); }); } } } } catch (error) { console.error("রেফারারকে স্পিন দিতে সমস্যা হয়েছে:", error); } }
